% Columns are drawn by recursion rather than a nested \loop, which TeX cannot nest.
\def\br@cells#1{%
  \ifnum\br@col>\br@cols\else
    \ifnum\br@cols=2 \def\br@w{\dimexpr(\linewidth-3.7em)/2\relax}\else\def\br@w{\dimexpr(\linewidth-5.2em)/3\relax}\fi
    \edef\br@id{#1.r\the\br@row.c\the\br@col}%
    \expandafter\field\expandafter{\br@id}{\br@w}%
    \ifnum\br@col<\br@cols\hspace{1.5em}\fi
    \advance\br@col 1
    \expandafter\br@cells\expandafter{#1}%
  \fi}
\newcommand{\blankrows}[3]{%
  \br@cols=#2\relax
  \br@row=1
  \loop\ifnum\br@row<\numexpr#3+1\relax
    \par\noindent\hspace*{2.2em}%
    \br@col=1 \br@cells{#1}%
    \par\vspace{0.2em}%
    \advance\br@row 1
  \repeat}
